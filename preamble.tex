% 中文支持 - 使用 ctex 包提供更好的中文排版
\usepackage[fontset=none]{ctex}

% 中文字体设置
% 正文：思源宋体 (Noto Serif CJK SC) — Google与Adobe联合开发，专为中文印刷排版设计
\setCJKmainfont{Noto Serif CJK SC}[
  BoldFont = Noto Serif CJK SC Bold,
]
% 标题/无衬线：思源黑体 (Noto Sans CJK SC)
\setCJKsansfont{Noto Sans CJK SC}[
  BoldFont = Noto Sans CJK SC Bold,
]
% 代码/等宽：思源等宽 (Noto Sans Mono CJK SC)
\setCJKmonofont{Noto Sans Mono CJK SC}

% 英文字体：正文/无衬线由CJK字体内置拉丁字符处理，确保中英文混排视觉大小一致
% 等宽字体需单独设置，否则代码块中拉丁字符回退到默认字体，与CJK等宽不一致
\setmonofont{Noto Sans Mono}[
  BoldFont = Noto Sans Mono Bold,
  Scale = MatchLowercase,
]

% 页面设置
\usepackage[top=2cm, bottom=2cm, left=2.5cm, right=2.5cm]{geometry}

% 行间距由 index.Rmd 的 linestretch 控制，不在此重复设置

% 段落格式：中文标准 — 仅段首缩进，段间不加额外间距
\usepackage{indentfirst}
\setlength{\parindent}{2em}   % 段首缩进两格（中文标准）
\setlength{\parskip}{0pt}     % 取消段间额外间距（覆盖pandoc模板的6pt）

% 数学支持
\usepackage{amsmath,amssymb,amsthm}
\usepackage{amsbsy}
\usepackage{booktabs}
\usepackage{longtable}
\makeatletter
\def\thm@space@setup{%
  \thm@preskip=8pt plus 2pt minus 4pt
  \thm@postskip=\thm@preskip
}
\makeatother


% 图表标题 - ctex 已优化中文标题
\usepackage{caption}
\captionsetup{font=small,labelsep=quad}

% 超链接 - 合并所有选项到单个配置
\usepackage[unicode]{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=blue,
  filecolor=magenta,
  urlcolor=cyan,
  pdftitle={AI赋能的生态统计},
  pdfauthor={沈国春、李勤},
  bookmarks=true,
  bookmarksopen=true,
  bookmarksnumbered=true,
  pdfencoding=auto
}

% 解决重复标签问题
\usepackage[all]{hypcap}

% 防止重复标签警告
\usepackage{cleveref}
\crefname{figure}{图}{图}
\crefname{table}{表}{表}
\crefname{equation}{公式}{公式}

% 代码环境 - R代码块使用Verbatim和Shaded环境
\usepackage{listings}
\lstset{
  basicstyle=\footnotesize\ttfamily,
  breaklines=true,
  frame=single,
  numbers=left,
  numberstyle=\tiny,
  stepnumber=1,
  numbersep=5pt,
  lineskip=-1pt
}

% 调整R代码块(Shaded/Highlighting/verbatim)的行间距
\usepackage{etoolbox}
\makeatletter
\AtBeginEnvironment{Shaded}{%
  \setstretch{1.0}%  % R代码块行间距，与正文一致
}
\AtBeginEnvironment{Highlighting}{%
  \setlength{\baselineskip}{1.0\baselineskip}%
}
\AtBeginEnvironment{verbatim}{%
  \setstretch{1.0}%  % R代码输出行间距，与正文一致
}
\makeatother

% AI协作面板 - 重定义quote环境，添加左侧蓝色边框
\usepackage{xcolor}
\usepackage{framed}
\definecolor{quote-bg}{HTML}{F0F4FA}
\definecolor{quote-border}{HTML}{5B9BD5}
\renewenvironment{quote}{%
  \def\FrameCommand{{\color{quote-border}\vrule width 4pt}\hspace{10pt}}%
  \MakeFramed {\advance\hsize-\width \FrameRestore\noindent\ignorespaces}%
}{%
  \endMakeFramed
}
